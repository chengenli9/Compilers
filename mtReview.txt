//================== challenge 4 =====================

abstract class Tree
{
   int visit(Visitor V);
}

class Leaf extends Tree
{
   int val;
   int visit(Vistor v);
}

class Branch extends Tree
{
   int val;
   Tree left;
   Tree right;

   int visit(Visitor v)
}

class MaxVisitor extends Visitor
{
   int visit(Branch b)
   {
      int val = b.val;
      int lval = b.left.accept(this); // call accept to do recursion
      int rval = b.right.accept(this);
      return Math.max(val, Math.max(lval, rval));
   }

   int visit(Leaf l)
   {
      return l.val;
   }

}
